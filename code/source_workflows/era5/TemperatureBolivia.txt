// BOLIVIA — MONTHLY TEMPERATURE PANEL
// ERA5-Land Monthly Aggregated | 1990-01 to 2026-05
// Geography: 339 municipalities, 2015 reference
// Source: ECMWF/ERA5_LAND/MONTHLY_AGGR
// Output: Annual CSV files, monthly municipality observations
// ============================================================================

// 1. SETTINGS

var ASSET = 'projects/ee-faviolc2/assets/bolivia339_2015';

var START_YEAR = 1990;
var START_MONTH = 1;
var END_YEAR = 2026;
var END_MONTH = 5;

// Run SMOKE_TEST first, then change to PRODUCTION.
var EXPORT_MODE = 'PRODUCTION';

var DRIVE_FOLDER = 'GEE_Bolivia_ERA5Land_Temperature_Monthly';

var SCALE = 11132;
var TILE_SCALE = 8;
var MAX_PIXELS_PER_REGION = 1e8;


// 2. MUNICIPALITIES

var municipalities = ee.FeatureCollection(ASSET).map(function(f) {
  f = ee.Feature(f);

  return f.set({
    mun_code: ee.String('000000')
      .cat(ee.String(f.get('mun_code'))).slice(-6),

    cna_code: ee.String('000000')
      .cat(ee.String(f.get('cna_code'))).slice(-6),

    municipality_area_m2: f.geometry().area(1)
  });
});

print('Municipalities:', municipalities.size());
print('Distinct mun_code:',
  municipalities.aggregate_count_distinct('mun_code'));
print('Distinct cna_code:',
  municipalities.aggregate_count_distinct('cna_code'));
print('First municipality:', municipalities.first());

Map.centerObject(municipalities, 6);
Map.addLayer(
  municipalities.style({
    color: 'FFFFFF',
    fillColor: '00000000',
    width: 1
  }),
  {},
  'Bolivia municipalities',
  true
);


// 3. ERA5-LAND

var era5 = ee.ImageCollection(
  'ECMWF/ERA5_LAND/MONTHLY_AGGR'
).select('temperature_2m');

print('ERA5 first image:', era5.first());
print('ERA5 available end:',
  ee.Date(era5.aggregate_max('system:time_start')));


// 4. SPATIAL REDUCER

var reducer = ee.Reducer.mean()
  .combine({
    reducer2: ee.Reducer.median(),
    sharedInputs: true
  })
  .combine({
    reducer2: ee.Reducer.minMax(),
    sharedInputs: true
  })
  .combine({
    reducer2: ee.Reducer.stdDev(),
    sharedInputs: true
  })
  .combine({
    reducer2: ee.Reducer.percentile([10, 25, 75, 90]),
    sharedInputs: true
  })
  .combine({
    reducer2: ee.Reducer.count(),
    sharedInputs: true
  });


// 5. NULL-SAFE FUNCTIONS

function safeDifference(a, b) {
  return ee.Algorithms.If(
    ee.Algorithms.IsEqual(a, null),
    null,
    ee.Algorithms.If(
      ee.Algorithms.IsEqual(b, null),
      null,
      ee.Number(a).subtract(ee.Number(b))
    )
  );
}

function safeDivide(a, b) {
  return ee.Algorithms.If(
    ee.Algorithms.IsEqual(a, null),
    null,
    ee.Algorithms.If(
      ee.Algorithms.IsEqual(b, null),
      null,
      ee.Algorithms.If(
        ee.Number(b).gt(0),
        ee.Number(a).divide(ee.Number(b)),
        null
      )
    )
  );
}


// 6. MONTHLY TEMPERATURE IMAGE

function monthlyImage(year, month) {

  year = ee.Number(year);
  month = ee.Number(month);

  var start = ee.Date.fromYMD(year, month, 1);
  var end = start.advance(1, 'month');

  var image = ee.Image(
    era5.filterDate(start, end).first()
  );

  return image
    .subtract(273.15)
    .rename('temperature')
    .toFloat()
    .set({
      year: year,
      month: month,
      ym: year.format('%04d')
        .cat('-')
        .cat(month.format('%02d')),
      'system:time_start': start.millis()
    });
}


// 7. MONTHLY MUNICIPAL STATISTICS

function monthlyStats(year, month) {

  year = ee.Number(year);
  month = ee.Number(month);

  var image = monthlyImage(year, month);

  // Temperature distribution
  var stats = image.reduceRegions({
    collection: municipalities,
    reducer: reducer,
    scale: SCALE,
    tileScale: TILE_SCALE,
    maxPixelsPerRegion: MAX_PIXELS_PER_REGION
  });

  // Observed area
  var areaImage = ee.Image.pixelArea()
    .updateMask(image.mask())
    .rename('valid_area');

  var areaStats = areaImage.reduceRegions({
    collection: municipalities,
    reducer: ee.Reducer.sum(),
    scale: SCALE,
    tileScale: TILE_SCALE,
    maxPixelsPerRegion: MAX_PIXELS_PER_REGION
  });

  // Join on municipal code
  var joined = ee.Join.saveFirst('area_feature').apply({
    primary: stats,
    secondary: areaStats,
    condition: ee.Filter.equals({
      leftField: 'mun_code',
      rightField: 'mun_code'
    })
  });

  var completed = ee.FeatureCollection(joined).map(function(f) {

    f = ee.Feature(f);

    var areaFeature = ee.Feature(f.get('area_feature'));

    // Single-band reducers produce mean, p75, count, sum, etc.
    var validArea = areaFeature.get('sum');
    var municipalityArea = f.get('municipality_area_m2');

    return ee.Feature(null, {
      mun_code: f.get('mun_code'),
      cna_code: f.get('cna_code'),
      municipio: f.get('municipio'),
      depto: f.get('depto'),
      provincia: f.get('provincia'),
      map_name: f.get('map_name'),
      map_year: f.get('map_year'),
      census_yr: f.get('census_yr'),

      year: year,
      month: month,
      ym: year.format('%04d')
        .cat('-')
        .cat(month.format('%02d')),
      date: ee.Date.fromYMD(year, month, 1)
        .format('YYYY-MM-dd'),

      temperature_mean: f.get('mean'),
      temperature_median: f.get('median'),
      temperature_min: f.get('min'),
      temperature_max: f.get('max'),
      temperature_stddev: f.get('stdDev'),
      temperature_p10: f.get('p10'),
      temperature_p25: f.get('p25'),
      temperature_p75: f.get('p75'),
      temperature_p90: f.get('p90'),

      temperature_iqr: safeDifference(
        f.get('p75'),
        f.get('p25')
      ),

      temperature_valid_pixel_count: f.get('count'),
      temperature_valid_area_m2: validArea,
      temperature_valid_area_share: safeDivide(
        validArea,
        municipalityArea
      ),

      municipality_area_m2: municipalityArea,

      product: 'ECMWF/ERA5_LAND/MONTHLY_AGGR',
      source_band: 'temperature_2m',
      source_units: 'kelvin',
      output_units: 'degrees_celsius',
      source_frequency: 'monthly',
      monthly_method: 'native_monthly_mean_of_hourly_temperature',
      temperature_definition: 'air_temperature_at_2_metres',
      conversion: 'temperature_2m - 273.15',
      scale_m: SCALE
    });
  });

  return completed;
}


// 8. ANNUAL TABLE

function tableForYear(year, firstMonth, lastMonth) {

  var months = ee.List.sequence(firstMonth, lastMonth);

  var tables = months.map(function(month) {
    return monthlyStats(year, month);
  });

  return ee.FeatureCollection(tables).flatten();
}

// 10. EXPORTS

var years = [];

if (EXPORT_MODE === 'SMOKE_TEST') {

  years = [2017];

} else if (EXPORT_MODE === 'PRODUCTION') {

  for (var y = START_YEAR; y <= END_YEAR; y++) {
    years.push(y);
  }

} else {
  throw new Error('Invalid EXPORT_MODE');
}

years.forEach(function(year) {

  var firstMonth = 1;
  var lastMonth = 12;

  if (year === START_YEAR) {
    firstMonth = START_MONTH;
  }

  if (year === END_YEAR) {
    lastMonth = END_MONTH;
  }

  if (EXPORT_MODE === 'SMOKE_TEST') {
    firstMonth = 8;
    lastMonth = 8;
  }

  var annualTable = tableForYear(
    year,
    firstMonth,
    lastMonth
  );

  var name = 'Bolivia_ERA5Land_Temperature_Monthly_Municipalities_' +
    String(year);

  Export.table.toDrive({
    collection: annualTable,
    description: name,
    folder: DRIVE_FOLDER,
    fileNamePrefix: name,
    fileFormat: 'CSV'
  });

});
